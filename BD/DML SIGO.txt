-- =========================================
-- Evita duplicados insertando solo los que no existan
WITH y AS (
  SELECT generate_series(2022, 2028) AS anio
),
q AS (
  SELECT 1 AS numero UNION ALL SELECT 2 UNION ALL SELECT 3
)
INSERT INTO periodo (anio, numero, estado)
SELECT y.anio, q.numero, TRUE
FROM y CROSS JOIN q
WHERE NOT EXISTS (
  SELECT 1 FROM periodo p WHERE p.anio = y.anio AND p.numero = q.numero
);
