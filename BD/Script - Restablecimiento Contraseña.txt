CREATE TABLE IF NOT EXISTS universidad.password_reset_token (
    token_id UUID PRIMARY KEY,
    usuario_id INT NOT NULL,
    secret_hash TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL,
    expires_at TIMESTAMPTZ NOT NULL,
    used_at TIMESTAMPTZ NULL,

    CONSTRAINT fk_password_reset_token_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES universidad.usuario (usuario_id)
        ON DELETE CASCADE
);

-- Índices
CREATE INDEX IF NOT EXISTS ix_password_reset_token_usuario_id
    ON universidad.password_reset_token (usuario_id);

CREATE INDEX IF NOT EXISTS ix_password_reset_token_expires_at
    ON universidad.password_reset_token (expires_at);


-- Tabla de OTP para reseteo
CREATE TABLE IF NOT EXISTS universidad.password_reset_otp (
    password_reset_otp_id SERIAL PRIMARY KEY,
    usuario_id INT NOT NULL,
    otp_hash TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL,
    expires_at TIMESTAMPTZ NOT NULL,
    attempts INT NOT NULL DEFAULT 0,
    used_at TIMESTAMPTZ NULL,
    last_sent_at TIMESTAMPTZ NULL,

    CONSTRAINT fk_password_reset_otp_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES universidad.usuario (usuario_id)
        ON DELETE CASCADE
);

-- Índices
CREATE INDEX IF NOT EXISTS ix_password_reset_otp_usuario_id
    ON universidad.password_reset_otp (usuario_id);

CREATE INDEX IF NOT EXISTS ix_password_reset_otp_expires_at
    ON universidad.password_reset_otp (expires_at);
