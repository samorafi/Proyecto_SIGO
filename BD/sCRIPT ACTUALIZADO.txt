-- ========================================================================================
-- Tablas con necesidad de CRUD
-- ========================================================================================

CREATE TABLE periodo (
  periodo_id SERIAL PRIMARY KEY,
  anio INT NOT NULL,
  numero INT NOT NULL CHECK (numero BETWEEN 1 AND 3),
  estado BOOLEAN NOT NULL DEFAULT TRUE
);

-- ============================================
-- Sección docentes
-- ============================================

CREATE TABLE persona (
  persona_id SERIAL PRIMARY KEY,
  nombre TEXT NOT NULL,
  cedula TEXT UNIQUE NOT NULL,
  genero_id INT REFERENCES genero(genero_id),
  correo TEXT UNIQUE NOT NULL,
  telefono TEXT,
  atestado_id INT REFERENCES atestado(atestado_id),
  provincia_id INT REFERENCES provincia(provincia_id),
  canton_id INT REFERENCES canton(canton_id),
  categoria_id INT REFERENCES categoria_docente(categoria_id),
  rol_docente_id INT REFERENCES rol_docente(rol_id),
  estado_persona_id INT REFERENCES estado_persona(estado_persona_id),
  fecha_ingreso DATE,
  tipo_contrato_id INT REFERENCES tipo_contrato(tipo_contrato_id),
  motivo_desvinculacion_id INT REFERENCES motivo_desvinculacion(motivo_id),
  periodo_desvinculacion_id INT REFERENCES periodo(periodo_id),
  comentarios TEXT
);

ALTER TABLE universidad.persona
  ALTER COLUMN estado_persona_id SET DEFAULT 1;

-- ============================================
-- Sección Académica
-- ============================================

CREATE TABLE carrera (
  carrera_id SERIAL PRIMARY KEY,
  nombre TEXT NOT NULL,  
  estado BOOLEAN NOT NULL DEFAULT TRUE,
  UNIQUE (nombre)
);

CREATE TABLE curso (
  curso_id SERIAL PRIMARY KEY,
  codigo TEXT NOT NULL,
  nombre TEXT NOT NULL, 
  carrera_id INT REFERENCES carrera(carrera_id),
  grado_id INT REFERENCES grado(grado_id),
  es_netcad BOOLEAN NOT NULL DEFAULT FALSE,
  estado BOOLEAN NOT NULL DEFAULT TRUE,
  UNIQUE (codigo)
);

-- ============================================
-- Sección de Coordinación
-- ============================================

CREATE TABLE coordinacion (
  coordinacion_id SERIAL PRIMARY KEY,
  persona_id INT NOT NULL REFERENCES persona(persona_id),
  carrera_id INT REFERENCES carrera(carrera_id),
  periodo_id INT NOT NULL REFERENCES periodo(periodo_id),
  estado BOOLEAN NOT NULL DEFAULT TRUE,
  comentarios TEXT
);

CREATE TABLE coordinacion_curso (
  coordinacion_curso_id SERIAL PRIMARY KEY,
  coordinacion_id INT NOT NULL REFERENCES universidad.coordinacion(coordinacion_id) ON DELETE CASCADE,
  curso_id INT NOT NULL REFERENCES universidad.curso(curso_id),
  estado BOOLEAN NOT NULL DEFAULT TRUE,
  comentarios TEXT
);

-- ============================================
-- Sección de Oferta Académica
-- ============================================

CREATE TABLE oferta (
  oferta_id SERIAL UNIQUE PRIMARY KEY,
  curso_id INT NOT NULL REFERENCES curso(curso_id),
  sede_id INT NOT NULL REFERENCES sede(sede_id), --
  modalidad_id INT NOT NULL REFERENCES modalidad(modalidad_id), --
  horario_id INT NOT NULL REFERENCES horario(horario_id), --
  periodo_id INT NOT NULL REFERENCES periodo(periodo_id), --
  accion_id INT REFERENCES accion_oferta(accion_id), --
  coordinador_id INT REFERENCES persona(persona_id), --
  comentarios TEXT
);

CREATE TABLE grupo (
  grupo_id SERIAL PRIMARY KEY,
  oferta_id INT NOT NULL REFERENCES oferta(oferta_id) ON DELETE CASCADE,
  numero INT NOT NULL,        
  cupo INT,
  matricula INT,
  UNIQUE (oferta_id, numero)
);

CREATE TABLE grupo_asignacion (
  grupo_asignacion_id SERIAL PRIMARY KEY,
  grupo_id INT NOT NULL REFERENCES grupo(grupo_id) ON DELETE CASCADE,
  persona_id INT NOT NULL REFERENCES persona(persona_id),
  rol_id INT NOT NULL REFERENCES rol_docente(rol_id),
  UNIQUE (grupo_id, persona_id, rol_id)
);

-- ============================================
-- Sección de Usuario, Roles de Usuario y Permisos
-- ============================================

CREATE TABLE usuario (
  usuario_id SERIAL PRIMARY KEY,
  nombre TEXT NOT NULL,
  correo TEXT UNIQUE NOT NULL,
  contrasena TEXT NOT NULL,
  activo BOOLEAN DEFAULT TRUE
);

ALTER TABLE universidad.usuario ADD COLUMN "AccessFailedCount" INTEGER NOT NULL DEFAULT 0;

-- 3. Habilitación de bloqueo (OPCIONAL: Para que el administrador lo gestione)

CREATE TABLE rol (
  rol_id SERIAL PRIMARY KEY,
  nombre TEXT UNIQUE NOT NULL,
  isSystem boolean
);

CREATE TABLE permiso (
  permiso_id SERIAL PRIMARY KEY,
  nombre TEXT NOT NULL,
  clave TEXT UNIQUE NOT NULL,
  ruta TEXT NOT NULL
);

CREATE TABLE usuario_rol ( -- RELACIÓN USUARIO - ROL (N:M)
  usuario_id INT NOT NULL REFERENCES usuario(usuario_id) ON DELETE CASCADE,
  rol_id INT NOT NULL REFERENCES rol(rol_id) ON DELETE CASCADE,
  PRIMARY KEY (usuario_id, rol_id)
);

CREATE TABLE rol_permiso ( -- RELACIÓN ROL - PERMISO (N:M)
  rol_id INT NOT NULL REFERENCES rol(rol_id) ON DELETE CASCADE,
  permiso_id INT NOT NULL REFERENCES permiso(permiso_id) ON DELETE CASCADE,
  PRIMARY KEY (rol_id, permiso_id)
);

-- 1) Agregamos columna tipo (C/T/P), por defecto 'C' para no romper lo existente
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM information_schema.columns
    WHERE table_schema='universidad' AND table_name='periodo' AND column_name='tipo'
  ) THEN
    ALTER TABLE periodo ADD COLUMN tipo char(1) NOT NULL DEFAULT 'C';
  END IF;
END $$;

-- 2) Generamos columna etiqueta para extraer la concatenacion de los datos
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM information_schema.columns
    WHERE table_schema='universidad' AND table_name='periodo' AND column_name='etiqueta'
  ) THEN
    ALTER TABLE periodo
    ADD COLUMN etiqueta text GENERATED ALWAYS AS (
      CASE tipo
        WHEN 'C' THEN (numero::text || 'C, ' || anio::text)
        WHEN 'T' THEN (numero::text || 'T, ' || anio::text)
        WHEN 'P' THEN (numero::text || 'P, ' || anio::text)
        ELSE (numero::text || ', ' || anio::text)
      END
    ) STORED;
  END IF;
END $$;

-- 3) Checks de rango por tipo
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint WHERE conname = 'ck_periodo_rangos_por_tipo'
  ) THEN
    ALTER TABLE periodo ADD CONSTRAINT ck_periodo_rangos_por_tipo
    CHECK (
      (tipo = 'C' AND numero BETWEEN 1 AND 3) OR
      (tipo = 'T' AND numero BETWEEN 1 AND 4) OR
      (tipo = 'P' AND numero BETWEEN 1 AND 12)
    );
  END IF;
END $$;

-- 4) Índice único por (año, tipo, número)
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_indexes WHERE indexname = 'ux_periodo_anio_tipo_numero'
  ) THEN
    CREATE UNIQUE INDEX ux_periodo_anio_tipo_numero
      ON periodo(anio, tipo, numero);
  END IF;
END $$;

-- 5) Migrar periodos existentes a tipo 'C' explícitamente (si todavía hay NULLs residuales)
UPDATE periodo SET tipo = 'C' WHERE tipo IS DISTINCT FROM 'C' AND tipo IS DISTINCT FROM 'T' AND tipo IS DISTINCT FROM 'P';

CREATE FUNCTION universidad.migrate_persona_periodo_ingreso_cuatrimestre(
  p_estado_default boolean DEFAULT true,               
  p_overwrite_existing boolean DEFAULT true           
)
RETURNS integer
LANGUAGE plpgsql
AS $$
DECLARE
  r                RECORD;
  v_anio           int;
  v_mes            int;
  v_numero         int;    
  v_periodo_id     int;
  v_processed      int := 0;
BEGIN
  -- Itera por todas las personas con fecha_ingreso (y opcionalmente sin periodo asignado)
  FOR r IN
    SELECT
      p.persona_id,                 
      p.fecha_ingreso,
      p.periodo_ingreso_id
    FROM universidad.persona p
    WHERE p.fecha_ingreso IS NOT NULL
      AND (p_overwrite_existing OR p.periodo_ingreso_id IS NULL)
  LOOP
    v_anio := EXTRACT(YEAR FROM r.fecha_ingreso)::int;
    v_mes  := EXTRACT(MONTH FROM r.fecha_ingreso)::int;

    -- Mapear mes -> cuatrimestre
    IF v_mes BETWEEN 1 AND 4 THEN
      v_numero := 1;       
    ELSIF v_mes BETWEEN 5 AND 8 THEN
      v_numero := 2;        
    ELSE
      v_numero := 3;      
    END IF;

    -- Upsert del periodo (tipo 'C')
    INSERT INTO universidad.periodo (anio, numero, tipo, estado)
    VALUES (v_anio, v_numero, 'C', p_estado_default)
    ON CONFLICT (anio, tipo, numero) DO UPDATE
      SET estado = universidad.periodo.estado
    RETURNING periodo_id INTO v_periodo_id;

    -- Asignar al registro de persona
    UPDATE universidad.persona
       SET periodo_ingreso_id = v_periodo_id
     WHERE persona_id = r.persona_id;

    v_processed := v_processed + 1;
  END LOOP;

  RETURN v_processed;
END;
$$;

-- EJECUTAR EL SIGUIENTE SELECT PARA APLICAR LOS CAMBIOS DE LA FUNCION --

SELECT universidad.migrate_persona_periodo_ingreso_cuatrimestre(
  p_estado_default := true,        
  p_overwrite_existing := false     
);

-- SP archivar_ofertas_por_periodo v1 - NO CUENTA CON MODALIDAD
CREATE OR REPLACE FUNCTION universidad.archivar_ofertas_por_periodo(
    p_periodo_id INT
)
RETURNS TEXT AS
$$
DECLARE
    v_total_archivadas INT;
    v_total_existentes INT;
BEGIN
    -- Contamos cuántas ofertas existen para ese período
    SELECT COUNT(*)
    INTO v_total_existentes
    FROM universidad.oferta
    WHERE periodo_id = p_periodo_id;

    IF v_total_existentes = 0 THEN
        RETURN 'No existen ofertas para el período especificado.';
    END IF;

    -- Archivamos solo las que no estén archivadas
    UPDATE universidad.oferta
    SET archivado = TRUE
    WHERE periodo_id = p_periodo_id
      AND (archivado IS NULL OR archivado = FALSE);

    GET DIAGNOSTICS v_total_archivadas = ROW_COUNT;

    -- Caso: ya estaban archivadas
    IF v_total_archivadas = 0 THEN
        RETURN 'Todas las ofertas de este período ya se encuentran archivadas.';
    END IF;

    -- Caso: se archivó algo
    RETURN FORMAT('Se archivaron %s oferta(s) del período.', v_total_archivadas);
END;
$$ LANGUAGE plpgsql;


-- SP archivar_ofertas_por_periodo v2 - CUENTA CON MODALIDAD
CREATE OR REPLACE FUNCTION universidad.archivar_ofertas_por_periodo_y_modalidad(
    p_periodo_id INT,
    p_modalidades INT[]
)
RETURNS TEXT AS
$$
DECLARE
    v_total_existentes INT;
    v_total_archivadas INT;
BEGIN
    -- Validar si existen ofertas con ese período y esas modalidades
    SELECT COUNT(*)
    INTO v_total_existentes
    FROM universidad.oferta
    WHERE periodo_id = p_periodo_id
      AND modalidad_id = ANY(p_modalidades);

    IF v_total_existentes = 0 THEN
        RETURN 'No existen ofertas para el período.';
    END IF;

    -- Actualizar (archivar)
    UPDATE universidad.oferta
    SET archivado = TRUE
    WHERE periodo_id = p_periodo_id
      AND modalidad_id = ANY(p_modalidades)
      AND (archivado IS NULL OR archivado = FALSE);

    GET DIAGNOSTICS v_total_archivadas = ROW_COUNT;

    IF v_total_archivadas = 0 THEN
        RETURN 'Todas las ofertas ya estaban archivadas para esta modalidad.';
    END IF;

    RETURN FORMAT(
        'Se archivaron %s oferta(s) del período %s para modalidades %s.',
        v_total_archivadas,
        p_periodo_id,
        array_to_string(p_modalidades, ', ')
    );
END;
$$ LANGUAGE plpgsql;


-- SP universidad.duplicar_ofertas_por_periodo

CREATE OR REPLACE FUNCTION universidad.duplicar_ofertas_por_periodo(
    p_periodo_origen INT,
    p_periodo_destino INT,
    p_modalidades INT[]
)
RETURNS TABLE (
    total_duplicadas INT,
    mensaje TEXT
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_count INT := 0;
BEGIN
    -- Validar si existen ofertas para duplicar
    SELECT COUNT(*) INTO v_count
    FROM universidad.oferta
    WHERE periodo_id = p_periodo_origen
      AND modalidad_id = ANY(p_modalidades);

    IF v_count = 0 THEN
        RETURN QUERY
        SELECT 0, 'No existen ofertas para duplicar con los criterios indicados.';
        RETURN;
    END IF;

    -- Insertar las ofertas duplicadas
    INSERT INTO universidad.oferta (
        curso_id,
        sede_id,
        modalidad_id,
        horario_id,
        periodo_id,
        accion_id,
        coordinador_id,
        comentarios,
        estado_oferta_id,
        grupo,
        cupo,
        matriculados,
        archivado
    )
    SELECT
        curso_id,
        sede_id,
        modalidad_id,
        horario_id,
        p_periodo_destino,   -- Nuevo periodo
        1,                   -- accion_id por defecto
        coordinador_id,
        '',                  -- comentarios vacíos
        2,                   -- estado_oferta_id por defecto
        grupo,
        cupo,
        0,                   -- matriculados reinicia
        FALSE                -- archivado siempre false
    FROM universidad.oferta
    WHERE periodo_id = p_periodo_origen
      AND modalidad_id = ANY(p_modalidades);

    GET DIAGNOSTICS v_count = ROW_COUNT;

    RETURN QUERY
